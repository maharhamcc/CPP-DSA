class Solution {
public:
    bool isValid(string s) {
        stack<char>st;
        bool flag=true;
        for(char c:s)
        {
            if(c=='(' || c=='{' || c=='[')
            {
                st.push(c);
            }
            else
            {
                if(st.empty())
                {
                    flag=false;
                    break;
                }
                else
                {
                    if(c==')' && st.top()=='(')
                    st.pop();
                    else if(c=='}' && st.top()=='{')
                    st.pop();
                    else if(c==']' && st.top()=='[')
                    st.pop();
                    else
                    {
                        flag=false;
                        break;
                    }
                }
            }
        }
        if(!st.empty())
        flag=false;
        return flag;

    }
};
